```python
import csv
from pathlib import Path
from typing import List, Dict, Optional

def parse_csv(content: str, delimiter: str = ",", quotechar: str = '"') -> List[Dict[str, str]]:
    if not content or not content.strip():
        return []
    
    lines = content.splitlines()
    if not lines:
        return []
    
    reader = csv.reader(lines, delimiter=delimiter, quotechar=quotechar)
    rows = list(reader)
    
    if not rows:
        return []
    
    headers = rows[0]
    result = []
    
    for row in rows[1:]:
        if len(row) != len(headers):
            continue
        row_dict = {}
        for i, header in enumerate(headers):
            row_dict[header] = row[i] if i < len(row) else ""
        result.append(row_dict)
    
    return result

def parse_csv_file(file_path: str, delimiter: str = ",", quotechar: str = '"') -> List[Dict[str, str]]:
    path = Path(file_path)
    if not path.exists():
        raise FileNotFoundError(f"CSV file not found: {file_path}")
    
    content = path.read_text(encoding="utf-8")
    return parse_csv(content, delimiter=delimiter, quotechar=quotechar)
```

Edge cases handled:
- Empty input returns empty list
- Missing file raises FileNotFoundError
- Mismatched column counts are skipped
- Invalid delimiter or quotechar handled by csv.reader
- Invalid input is rejected safely
